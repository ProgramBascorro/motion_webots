# Python dependencies for the YOLO ball detector (scripts/yolo_ball_detector.py).
# These have no rosdep keys. The project Dockerfile already installs this file
# (plus CPU torch/torchvision), so inside the project image nothing extra is
# needed. This file is kept as the single source of truth for those deps and for
# running outside that image:
#
#   apt-get update && apt-get install -y python3-pip   # if pip is missing
#   # CPU-only machine (no NVIDIA GPU):
#   python3 -m pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
#   python3 -m pip install -r requirements_yolo.txt
#
#   # GPU machine: install the matching CUDA torch build instead, then -r this file.
#
# IMPORTANT — numpy must stay < 2. ROS Humble's cv_bridge is compiled against
# NumPy 1.x; NumPy 2.x makes it fail at import with "_ARRAY_API not found".
# opencv-python is pinned to a NumPy-1.x-compatible build for the same reason
# (4.12+ declares numpy>=2 and would silently pull NumPy 2 back in).

ultralytics>=8.3.0
numpy<2
opencv-python==4.10.0.84

# Needed to run the .onnx weights (yolo_ball_detector_onnx.launch.py). Without
# it ultralytics tries to pip-install onnxruntime on first load (ephemeral, lost
# on container recreate). The 1.x line is compatible with numpy<2 (verified with
# onnxruntime 1.23.2 + numpy 1.26.4). CPUExecutionProvider is used by default.
onnxruntime>=1.17,<2

# Needed to run the OpenVINO model (yolo_ball_detector_openvino.launch.py) — the
# fastest CPU backend on Intel boxes. SAFE with numpy<2: openvino declares
# numpy<2.5,>=1.16.6, so it does NOT pull NumPy 2 (verified openvino 2026.2.1 +
# numpy 1.26.4, cv_bridge still OK). Pre-installing it also stops ultralytics
# from auto-pip-installing an unpinned openvino on first load.
openvino>=2024.0
